import Button from './buttons';
import Field from './Field';

function App() {
  
  const headers = [
    { label: "lock", colSpan: 1 },
    { label: "Units", colSpan: 1 },
    { label: "Value", colSpan: 1 },
    { label: "Funds Actions", colSpan: 3 },
  ];

  const rows = [
    { chain: 'LBG Chain', units: '1m', value: '$5,000,000' },
    { chain: 'Hedera', units: '500k', value: '$150,000' },
    { chain: 'Canton', units: '250k', value: '$1,500,000' },

  return (
    <div className="bg-gray-200 fixed inset-0 flex items-center justify-center">
      {/* Updated container background to a soft eggshell white (#F0EAD6) */}
      <div className="bg-[#c9d8e9] p-8 rounded-lg shadow-lg flex flex-col items-center justify-center w-fit min-h-fit">
        <h1 className="text-xl font-bold text-black text-center mb-8">

        </h1>

        <div className="grid grid-cols-6 gap-x-2 w-full text-center text-black font-semibold mb-4">
          {headers.map((header, index) => {
            // For "Funds Actions", create a nested grid with 3 columns so that the label sits in the middle.
            if (header.label === "Funds Actions") {
              return (
                <div key={index} className="col-span-3 grid grid-cols-3">
                  <div></div> {/* Empty cell on the left */}

                {header.label}

          })}

}

export default App;
